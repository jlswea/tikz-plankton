Quick start

\documentclass[border=1cm]{standalone}
\usepackage{plankton-base}
\usepackage{plankton-electrical}
\usepackage{plankton-furniture}

\begin{document}
\begin{tikzpicture}[x=1cm, y=1cm]
  % Walls
  \planwall{0,0}{6,0}
  \planwall{6,0}{6,4}
  \planwall{6,4}{0,4}
  \planwall{0,4}{0,0}

  % Door and window
  \plandoor{1,0}{0}
  \planwindow{2,4}{4,4}

  % Electrical
  \planoutlet{0.2,1}{0}
  \planlight{3,2}{0}

  % Furniture
  \plansofa[seats=3]{0.3,1.5}{0}
  \plantable[shape=round]{3,1.5}{0.8}{0.8}

  % Label
  \planlabel{3,3.5}{Living Room}
\end{tikzpicture}
\end{document}
